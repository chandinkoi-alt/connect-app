const express = require('express'); const cors = require('cors'); const path = require('path'); const app = express(); const PORT = process.env.PORT || 3000; // Cấu hình Middleware app.use(cors()); app.use(express.json()); // Phục vụ các file tĩnh trong thư mục public (CSS, JS, hình ảnh...) app.use(express.static(path.join(__dirname, 'public'))); // Trả về thông tin API app.get('/api/info', (req, res) => { res.json({ unionName: process.env.UNION_NAME || 'コネクト協同組合', permitNo: process.env.PERMIT_NO || '許1808000401', status: 'Active' }); }); // Route kiểm tra sức khỏe hệ thống app.get('/health', (req, res) => { res.send('OK'); }); // BẮT BỘC: Mở file index.html trong thư mục public khi truy cập trang chủ app.get('/', (req, res) => { res.sendFile(path.join(__dirname, 'public', 'index.html')); }); app.listen(PORT, () => { console.log(`Server is running on port ${PORT}`); });
